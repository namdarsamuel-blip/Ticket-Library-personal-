The app now carries -- those along per event under their original column header instead of -- silently dropping them on import, but it needs somewhere to persist -- them: a new JSONB column, plus both save functions updated to actually -- write it (without this, uploads would show the extra columns in the -- browser for the current session only, then lose them on next reload -- since the column wouldn't exist to save them into). alter table events add column if not exists extra_fields jsonb default '{}'::jsonb; end if; if v_matched_id is not null then update events set sheet_mode = p_sheet_mode, name = coalesce(e->>'name', ''), venue = coalesce(e->>'venue', ''), date_str = coalesce(e->>'date_str', ''), primary_site = coalesce(e->>'primary_site', ''), primary_link = coalesce(e->>'primary_link', ''), vivid_link = coalesce(e->>'vivid_link', ''), stub_link = coalesce(e->>'stub_link', ''), note = coalesce(e->>'note', ''), rating = coalesce((e->>'rating')::int, 0), pinned = coalesce((e->>'pinned')::boolean, false), sections = coalesce(e->'sections', '[]'::jsonb), is_new = coalesce((e->>'is_new')::boolean, false), archived = coalesce((e->>'archived')::boolean, false), note_history = coalesce(e->'note_history', '[]'::jsonb), presale_date_str = coalesce(e->>'presale_date_str', ''), code = coalesce(e->>'code', ''), public_sale_date_str = coalesce(e->>'public_sale_date_str', ''), price_history = coalesce(e->'price_history', '[]'::jsonb), extra_links = coalesce(e->'extra_links', '[]'::jsonb), extra_fields = coalesce(e->'extra_fields', '{}'::jsonb) where id = v_matched_id;
